// Qualification-only transparent fixed-loopback TLS relay. It reports public
// ClientHello/ServerHello ALPN; it never decrypts or records application bytes.
package main

import (
	"encoding/binary"
	"encoding/json"
	"io"
	"net"
	"os"
	"os/signal"
	"sync"
	"syscall"
	"time"
)

var outputMu sync.Mutex

func emit(v any) { outputMu.Lock(); defer outputMu.Unlock(); json.NewEncoder(os.Stdout).Encode(v) }

type observer struct {
	records, handshake []byte
	done               bool
	direction          string
	id                 int
}

func (o *observer) feed(data []byte) {
	if o.done {
		return
	}
	if len(o.records)+len(o.handshake)+len(data) > 65536 {
		o.done = true
		return
	}
	o.records = append(o.records, data...)
	for len(o.records) >= 5 {
		n := int(binary.BigEndian.Uint16(o.records[3:5]))
		if n > 18432 {
			o.done = true
			return
		}
		if len(o.records) < 5+n {
			return
		}
		kind := o.records[0]
		if kind != 22 {
			o.done = true
			return
		}
		o.handshake = append(o.handshake, o.records[5:5+n]...)
		o.records = o.records[5+n:]
		if len(o.handshake) < 4 {
			continue
		}
		want := byte(1)
		if o.direction == "server" {
			want = 2
		}
		if o.handshake[0] != want {
			o.done = true
			return
		}
		length := int(o.handshake[1])<<16 | int(o.handshake[2])<<8 | int(o.handshake[3])
		if length > 65532 {
			o.done = true
			return
		}
		if len(o.handshake) < 4+length {
			continue
		}
		alpn, version, ok := parseHello(o.handshake[4:4+length], o.direction)
		if ok {
			emit(map[string]any{"event": "HELLO", "connection": o.id, "direction": o.direction, "alpn": alpn, "tlsVersion": version})
		}
		o.done = true
		o.records = nil
		o.handshake = nil
		return
	}
}

func parseHello(b []byte, direction string) ([]string, string, bool) {
	if len(b) < 35 {
		return nil, "", false
	}
	version := "legacy-unknown"
	if b[0] == 3 && b[1] == 3 {
		version = "1.2"
	}
	i := 35 + int(b[34])
	if i > len(b) {
		return nil, "", false
	}
	if direction == "client" {
		if i+2 > len(b) {
			return nil, "", false
		}
		n := int(binary.BigEndian.Uint16(b[i : i+2]))
		i += 2 + n
		if i >= len(b) {
			return nil, "", false
		}
		n = int(b[i])
		i += 1 + n
	} else {
		i += 3
	}
	if i+2 > len(b) {
		return nil, "", false
	}
	n := int(binary.BigEndian.Uint16(b[i : i+2]))
	i += 2
	end := i + n
	if end != len(b) {
		return nil, "", false
	}
	alpn := []string{}
	for i < end {
		if i+4 > end {
			return nil, "", false
		}
		kind := binary.BigEndian.Uint16(b[i : i+2])
		n := int(binary.BigEndian.Uint16(b[i+2 : i+4]))
		i += 4
		if i+n > end {
			return nil, "", false
		}
		data := b[i : i+n]
		i += n
		if kind == 43 && direction == "server" && len(data) == 2 && data[0] == 3 && data[1] == 4 {
			version = "1.3"
		}
		if kind != 16 {
			continue
		}
		if len(data) < 2 || int(binary.BigEndian.Uint16(data[:2])) != len(data)-2 {
			return nil, "", false
		}
		for p := 2; p < len(data); {
			length := int(data[p])
			p++
			if length == 0 || p+length > len(data) {
				return nil, "", false
			}
			value := string(data[p : p+length])
			p += length
			// Only recognized public protocol identifiers enter retained output.
			switch value {
			case "h2", "http/1.1", "h3":
				alpn = append(alpn, value)
			default:
				alpn = append(alpn, "OTHER")
			}
		}
	}
	return alpn, version, true
}

func relay(dst, src net.Conn, direction string, id int) {
	defer dst.Close()
	defer src.Close()
	o := observer{direction: direction, id: id}
	b := make([]byte, 32768)
	for {
		n, err := src.Read(b)
		if n > 0 {
			o.feed(b[:n])
			if _, e := io.CopyN(dst, newReader(b[:n]), int64(n)); e != nil {
				return
			}
		}
		if err != nil {
			return
		}
	}
}

type byteReader struct{ b []byte }

func newReader(b []byte) *byteReader { return &byteReader{b: b} }
func (r *byteReader) Read(p []byte) (int, error) {
	if len(r.b) == 0 {
		return 0, io.EOF
	}
	n := copy(p, r.b)
	r.b = r.b[n:]
	return n, nil
}

func main() {
	if (len(os.Args) != 2 && len(os.Args) != 3) || os.Args[1] != "observe" {
		os.Exit(2)
	}
	l, err := net.Listen("tcp", "127.0.0.1:24860")
	if err != nil {
		emit(map[string]any{"event": "FAIL", "code": "LOOPBACK_OBSERVER_LISTEN_FAILED"})
		os.Exit(1)
	}
	var mu sync.Mutex
	connections := map[net.Conn]struct{}{}
	var workers sync.WaitGroup
	stopping := false
	accepted := 0
	stop := func() {
		mu.Lock()
		defer mu.Unlock()
		if stopping {
			return
		}
		stopping = true
		l.Close()
		for c := range connections {
			c.Close()
		}
	}
	defer stop()
	if len(os.Args) == 3 && os.Args[2] == "--stdio-lifecycle" {
		go func() { _, _ = io.Copy(io.Discard, os.Stdin); stop() }()
	}
	signals := make(chan os.Signal, 1)
	signal.Notify(signals, syscall.SIGTERM, syscall.SIGINT)
	defer signal.Stop(signals)
	timer := time.NewTimer(180 * time.Second)
	defer timer.Stop()
	ended := make(chan struct{})
	go func() {
		select {
		case <-signals:
			stop()
		case <-timer.C:
			stop()
		case <-ended:
		}
	}()
	emit(map[string]any{"event": "READY", "pid": os.Getpid(), "loopback": true, "fixedTarget": true})
	for {
		client, err := l.Accept()
		if err != nil {
			break
		}
		mu.Lock()
		if stopping || len(connections) >= 16 {
			mu.Unlock()
			client.Close()
			continue
		}
		accepted++
		id := accepted
		connections[client] = struct{}{}
		workers.Add(1)
		mu.Unlock()
		go func() {
			defer workers.Done()
			defer client.Close()
			defer func() { mu.Lock(); delete(connections, client); mu.Unlock() }()
			server, err := net.DialTimeout("tcp", "127.0.0.1:24862", 3*time.Second)
			if err != nil {
				return
			}
			defer server.Close()
			mu.Lock()
			if stopping {
				mu.Unlock()
				return
			}
			connections[server] = struct{}{}
			mu.Unlock()
			defer func() { mu.Lock(); delete(connections, server); mu.Unlock() }()
			var pair sync.WaitGroup
			pair.Add(2)
			go func() { defer pair.Done(); relay(server, client, "client", id) }()
			go func() { defer pair.Done(); relay(client, server, "server", id) }()
			pair.Wait()
		}()
	}
	stop()
	workers.Wait()
	close(ended)
	emit(map[string]any{"event": "STOPPED", "accepted": accepted, "active": 0})
}
